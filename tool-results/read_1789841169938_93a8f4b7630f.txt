     1→/**
     2→ * APEX KART — Procedural kart + character meshes and code-driven animation.
     3→ * No external models: every racer is assembled from primitives with a
     4→ * species-unique head, and animated procedurally (steer lean, drift roll,
     5→ * landing bounce, victory/defeat poses, hit flash).
     6→ *
     7→ * Build strategy: all static parts are merged per material (chassis / accent /
     8→ * dark / tire / hub) with BufferGeometryUtils.mergeGeometries, so a full kart
     9→ * costs ~10 draw calls while looking like a hand-built party racer:
    10→ * sculpted monocoque + nose cone, side pods with intakes, wheel fenders,
    11→ * front wing with endplates, rear wing on struts, twin exhaust, roll hoop.
    12→ */

    53→const numberTexCache = new Map<string, THREE.CanvasTexture>();

    55→/** White disc with the racer's number — the classic karting chest badge. */
    56→function getNumberTexture(id: string): THREE.CanvasTexture {
    57→  const hit = numberTexCache.get(id);
    58→  if (hit) return hit;
    59→  const c = document.createElement('canvas');
    60→  c.width = c.height = 128;
    61→  const ctx = c.getContext('2d')!;
    62→  ctx.clearRect(0, 0, 128, 128);
    63→  ctx.fillStyle = '#f8f8ff';
    64→  ctx.beginPath(); ctx.arc(64, 64, 60, 0, Math.PI * 2); ctx.fill();
    65→  ctx.strokeStyle = '#1a1a28'; ctx.lineWidth = 8;
    66→  ctx.beginPath(); ctx.arc(64, 64, 54, 0, Math.PI * 2); ctx.stroke();
    67→  ctx.fillStyle = '#1a1a28';
    68→  ctx.font = '900 64px Arial, sans-serif';
    69→  ctx.textAlign = 'center';
    70→  ctx.textBaseline = 'middle';
    71→  ctx.fillText(String(RACE_NUMBERS[id] ?? 0), 64, 68);
    72→  const tex = new THREE.CanvasTexture(c);
    73→  numberTexCache.set(id, tex);
    74→  return tex;
    75→}

    77→/** Shared "APEX star" racing roundel decal (built once). */
    78→let roundelTex: THREE.CanvasTexture | null = null;
    79→function getRoundelTexture(): THREE.CanvasTexture {
    80→  if (roundelTex) return roundelTex;
    81→  const c = document.createElement('canvas');
    82→  c.width = c.height = 128;
    83→  const ctx = c.getContext('2d')!;
    84→  ctx.clearRect(0, 0, 128, 128);
    85→  ctx.fillStyle = '#f8f8ff';
    86→  ctx.beginPath(); ctx.arc(64, 64, 62, 0, Math.PI * 2); ctx.fill();
    87→  ctx.strokeStyle = '#1a1a28'; ctx.lineWidth = 7;
    88→  ctx.beginPath(); ctx.arc(64, 64, 57, 0, Math.PI * 2); ctx.stroke();
    89→  ctx.fillStyle = '#ff9a2a';
    90→  ctx.beginPath();
    91→  for (let i = 0; i < 10; i++) {
    92→    const r = i % 2 ? 17 : 42;
    93→    const a = -Math.PI / 2 + (i / 10) * Math.PI * 2;
    94→    const x = 64 + Math.cos(a) * r, y = 64 + Math.sin(a) * r;
    95→    if (i) ctx.lineTo(x, y); else ctx.moveTo(x, y);
    96→  }
    97→  ctx.closePath(); ctx.fill();
    98→  roundelTex = new THREE.CanvasTexture(c);
    99→  return roundelTex;
   100→}

   102→/** geometry transform helper: translate + rotate (Euler XYZ), returns geo */
   103→const T = (
   104→  geo: THREE.BufferGeometry,
   105→  x: number, y: number, z: number,
   106→  rx = 0, ry = 0, rz = 0,
   107→): THREE.BufferGeometry => {
   108→  const m = new THREE.Matrix4();
   109→  m.makeRotationFromEuler(new THREE.Euler(rx, ry, rz));
   110→  m.setPosition(x, y, z);
   111→  const g = geo.clone();
   112→  g.applyMatrix4(m);
   113→  return g;
   114→};

   150→  constructor(characterId: string, kartColor: number, models?: VisualModels) {
   151→    this.characterId = characterId;
   152→    const driverOwnsKart = !!models?.driver && !!models.driverIncludesKart;
   153→    if (models?.kart && !driverOwnsKart) {
   154→      this.buildKartGLB(models.kart, models.kartMeta, kartColor);
   155→    } else {
   156→      this.buildKart(kartColor);
   157→    }
   158→    if (models?.driver) {
   159→      this.buildCharacterGLB(models.driver, driverOwnsKart);
   160→    } else {
   161→      this.buildCharacter(characterId);
   162→    }
   163→    this.group.add(this.body);
   164→    if (!driverOwnsKart) this.body.add(this.characterRoot);
   165→  }

   169→  /** GLB chassis: adopt the model as the body, wire detected wheels + flames. */
   170→  private buildKartGLB(scene: THREE.Object3D, meta: KartModelMeta | undefined, color: number): void {
   171→    this.usedGlbKart = true;
   172→    scene.traverse(o => {
   173→      const m = o as THREE.Mesh;
   174→      if (m.isMesh && (m.material as THREE.MeshStandardMaterial)?.color) {
   175→        this.materials.push(m.material as THREE.MeshLambertMaterial);
   176→      }
   177→    });
   178→    // tint with the racer chassis color (per-instance materials, cloned upstream)
   179→    const tintables = meta?.tintables ?? [];
   180→    for (const mat of tintables) {
   181→      (mat as THREE.MeshStandardMaterial).color.setHex(color);
   182→      this.chassisMats.push(mat as THREE.MeshLambertMaterial);
   183→    }
   184→    this.body.add(scene);
   185→
   186→    this.glbWheels = meta?.wheels ?? [];
   187→    this.glbFrontWheels = meta?.frontWheels ?? [];
   188→    for (const w of this.glbFrontWheels) this.glbSteerBase.set(w, w.rotation.y);
   189→    this.glbRearZ = meta?.rearZ ?? -1.9;
   190→    if (meta?.seat) {
   191→      this.glbSeat = { x: meta.seat.x, y: meta.seat.y, z: meta.seat.z };
   192→      this.seatScale = meta.seat.scale ?? 1;
   193→      this.seatY = meta.seat.y;
   194→    }
   195→
   196→    // boost flames + emissive exhaust tips at the model's rear
   197→    const tipMat = new THREE.MeshLambertMaterial({ color: 0x8a8a96, emissive: 0x220000 });
   198→    for (const s of [-1, 1]) {
   199→      const flame = new THREE.Mesh(new THREE.ConeGeometry(0.17, 1.0, 6),
   200→        new THREE.MeshBasicMaterial({ color: 0xffa53a, transparent: true, opacity: 0.9 }));
   201→      flame.rotation.x = Math.PI / 2;
   202→      flame.position.set(s * 0.35, 0.55, this.glbRearZ - 0.55);
   203→      flame.visible = false;
   204→      this.body.add(flame);
   205→      this.boostFlames.push(flame);
   206→    }
   207→    this.materials.push(tipMat);
   208→
   209→    // No usable wheels IN the model → add the procedural wheels only
   210→    // (never the whole kart — the GLB is the chassis now) so the kart never
   211→    // looks frozen. Models that ship their own wheels skip these entirely.
   212→    if (this.glbWheels.length < 3 && !meta?.modelWheels) this.buildProceduralWheels();
   213→  }
   214→
   215→  /** Procedural wheel set — shared by the procedural kart and the GLB fallback. */
   216→  private buildProceduralWheels(): void {
   217→    const tireMat = new THREE.MeshLambertMaterial({ color: TIRE });
   218→    const hubMat = new THREE.MeshLambertMaterial({ color: 0xd8dce4 });
   219→    this.materials.push(tireMat, hubMat);
   220→
   221→    const makeWheel = (radius: number, width: number): { tire: THREE.Mesh; hub: THREE.Mesh } => {
   222→      const tireGeo = new THREE.CylinderGeometry(radius, radius, width, 14);
   223→      tireGeo.rotateZ(Math.PI / 2);
   224→      const tire = new THREE.Mesh(tireGeo, tireMat);
   225→      tire.castShadow = true;
   226→      const hubParts: THREE.BufferGeometry[] = [
   227→        T(new THREE.CylinderGeometry(radius * 0.52, radius * 0.52, width * 1.06, 10), 0, 0, 0, 0, 0, Math.PI / 2),
   228→      ];
   229→      for (let i = 0; i < 5; i++) {
   230→        const a = (i / 5) * Math.PI * 2;
   231→        hubParts.push(T(new THREE.BoxGeometry(width * 0.35, radius * 1.7, radius * 0.16), 0, 0, 0, a, 0, 0));
   232→      }
   233→      const hubGeo = mergeGeometries(hubParts);
   234→      const hub = new THREE.Mesh(hubGeo ?? new THREE.BoxGeometry(0.01, 0.01, 0.01), hubMat);
   235→      hub.castShadow = false;
   236→      return { tire, hub };
   237→    };
   238→
   239→    const wheelAt = (x: number, z: number, front: boolean, radius: number, width: number): void => {
   240→      const holder = new THREE.Group();
   241→      const { tire, hub } = makeWheel(radius, width);
   242→      holder.add(tire, hub);
   243→      holder.position.set(x, radius, z);
   244→      this.body.add(holder);
   245→      this.wheels.push(tire);
   246→      if (front) this.frontWheels.push(holder);
   247→    };
   248→    wheelAt(-0.94, 1.12, true, 0.4, 0.32);
   249→    wheelAt(0.94, 1.12, true, 0.4, 0.32);
   250→    wheelAt(-0.98, -1.02, false, 0.47, 0.42);
   251→    wheelAt(0.98, -1.02, false, 0.47, 0.42);
   252→  }

   254→  /** GLB driver: seated figure, or a full character+kart model as the body. */
   255→  private buildCharacterGLB(scene: THREE.Object3D, includesKart: boolean): void {
   256→    scene.traverse(o => {
   257→      const m = o as THREE.Mesh;
   258→      if (m.isMesh && (m.material as THREE.MeshStandardMaterial)?.color) {
   259→        this.materials.push(m.material as THREE.MeshLambertMaterial);
   260→      }
   261→    });
   262→    if (includesKart) {
   263→      this.usedGlbKart = true;
   264→      this.body.add(scene);
   265→    } else {
   266→      this.characterRoot.position.set(0, 0.6, -0.3);
   267→      this.torso.add(scene);
   268→      scene.position.set(0, 0.18, 0); // seat the figure on the cockpit floor
   269→    }
   270→  }

   285→  private buildKart(color: number): void {
   286→    const chassis = new THREE.MeshLambertMaterial({ color });
   287→    const accent = new THREE.MeshLambertMaterial({ color: ACCENT });
   288→    const dark = new THREE.MeshLambertMaterial({ color: DARK });
   289→    this.chassisMats.push(chassis);
   290→    this.materials.push(chassis, accent, dark);
   291→
   292→    // ===================== CHASSIS-COLORED PARTS (merged) =====================
   293→    const chassisParts: THREE.BufferGeometry[] = [];
   294→    // main monocoque tub — slim nose widening to the cockpit
   295→    chassisParts.push(T(new THREE.BoxGeometry(1.5, 0.4, 2.6), 0, 0.42, -0.1));
   296→    // rounded cowl over the cockpit front (squashed sphere)
   297→    const cowl = new THREE.SphereGeometry(0.62, 12, 10);
   298→    cowl.scale(1.15, 0.62, 1.25);
   299→    chassisParts.push(T(cowl, 0, 0.55, 0.55));
   300→    // nose cone (tapered cylinder pointing forward)
   301→    chassisParts.push(T(new THREE.CylinderGeometry(0.16, 0.58, 1.15, 6), 0, 0.42, 1.62, Math.PI / 2));
   302→    // side pods with raised shoulders
   303→    for (const s of [-1, 1]) {
   304→      chassisParts.push(T(new THREE.BoxGeometry(0.5, 0.36, 1.7), s * 0.95, 0.44, -0.25));
   305→      chassisParts.push(T(new THREE.BoxGeometry(0.54, 0.14, 0.5), s * 0.95, 0.66, 0.35));
   306→    }
   307→    // rear engine cowl + cooling fins
   308→    chassisParts.push(T(new THREE.BoxGeometry(1.05, 0.42, 0.85), 0, 0.62, -1.05));
   309→    for (let i = 0; i < 4; i++) {
   310→      chassisParts.push(T(new THREE.BoxGeometry(0.8, 0.045, 0.5), 0, 0.86 + i * 0.07, -1.1));
   311→    }
   312→    // fender planks over both axles
   313→    for (const s of [-1, 1]) {
   314→      chassisParts.push(T(new THREE.BoxGeometry(0.72, 0.14, 1.35), s * 0.98, 0.82, 1.05));
   315→      chassisParts.push(T(new THREE.BoxGeometry(0.78, 0.14, 1.45), s * 1.0, 0.82, -1.02));
   316→    }
   317→    // seat back
   318→    chassisParts.push(T(new THREE.BoxGeometry(0.82, 0.62, 0.2), 0, 0.98, -0.86, -0.12));
   319→
   320→    // ===================== ACCENT PARTS (merged) ==============================
   321→    const accentParts: THREE.BufferGeometry[] = [];
   322→    // front wing + endplates
   323→    accentParts.push(T(new THREE.BoxGeometry(1.72, 0.09, 0.55), 0, 0.3, 2.02, -0.12));
   324→    for (const s of [-1, 1]) {
   325→      accentParts.push(T(new THREE.BoxGeometry(0.09, 0.3, 0.6), s * 0.86, 0.4, 2.0));
   326→    }
   327→    // rear wing + struts + endplates
   328→    accentParts.push(T(new THREE.BoxGeometry(1.62, 0.09, 0.6), 0, 1.32, -1.62, 0.1));
   329→    for (const s of [-1, 1]) {
   330→      accentParts.push(T(new THREE.BoxGeometry(0.09, 0.36, 0.62), s * 0.8, 1.42, -1.62));
   331→      accentParts.push(T(new THREE.BoxGeometry(0.08, 0.5, 0.08), s * 0.5, 1.05, -1.6));
   332→    }
   333→    // racing stripe down the nose + cowl
   334→    accentParts.push(T(new THREE.BoxGeometry(0.3, 0.06, 1.9), 0, 0.645, 0.72));
   335→    // side-pod intake lips
   336→    for (const s of [-1, 1]) {
   337→      accentParts.push(T(new THREE.BoxGeometry(0.1, 0.2, 0.5), s * 1.21, 0.5, 0.42));
   338→    }
   339→    // roll hoop over the driver (half torus)
   340→    accentParts.push(T(new THREE.TorusGeometry(0.46, 0.07, 6, 10, Math.PI), 0, 1.05, -0.55, 0, Math.PI / 2));
   341→
   342→    // ===================== DARK PARTS (merged) ================================
   343→    const darkParts: THREE.BufferGeometry[] = [];
   344→    // floor pan
   345→    darkParts.push(T(new THREE.BoxGeometry(1.34, 0.12, 3.2), 0, 0.2, -0.05));
   346→    // cockpit opening (dark tub interior)
   347→    darkParts.push(T(new THREE.BoxGeometry(0.96, 0.16, 1.15), 0, 0.64, -0.12));
   348→    // front splitter + rear diffuser strakes
   349→    darkParts.push(T(new THREE.BoxGeometry(1.56, 0.08, 0.4), 0, 0.22, 2.12));
   350→    for (const s of [-1, 1]) {
   351→      darkParts.push(T(new THREE.BoxGeometry(0.08, 0.22, 0.5), s * 0.45, 0.2, -1.72));
   352→      darkParts.push(T(new THREE.BoxGeometry(0.08, 0.22, 0.5), s * 0.75, 0.2, -1.72));
   353→    }
   354→    // steering column + wheel
   355→    darkParts.push(T(new THREE.CylinderGeometry(0.05, 0.05, 0.5, 6), 0, 0.78, 0.42, 0.7));
   356→    darkParts.push(T(new THREE.TorusGeometry(0.2, 0.05, 6, 12), 0, 0.93, 0.5, 1.05));
   357→    // twin exhaust pipes
   358→    for (const s of [-1, 1]) {
   359→      darkParts.push(T(new THREE.CylinderGeometry(0.085, 0.1, 0.55, 7), s * 0.32, 0.56, -1.52, Math.PI / 2));
   360→    }
   361→    // side mirrors on stalks
   362→    for (const s of [-1, 1]) {
   363→      darkParts.push(T(new THREE.CylinderGeometry(0.022, 0.022, 0.3, 5), s * 0.88, 0.95, 0.55, 0, s * 0.9));
   364→      darkParts.push(T(new THREE.BoxGeometry(0.05, 0.12, 0.16), s * 0.99, 1.05, 0.58));
   365→    }
   366→
   367→    // ---- merge & add ----------------------------------------------------------
   368→    const addMerged = (parts: THREE.BufferGeometry[], material: THREE.Material): void => {
   369→      const merged = mergeGeometries(parts);
   370→      if (!merged) return;
   371→      const mesh = new THREE.Mesh(merged, material);
   372→      mesh.castShadow = true;
   373→      this.body.add(mesh);
   374→      for (const g of parts) g.dispose();
   375→    };
   376→    addMerged(chassisParts, chassis);
   377→    addMerged(accentParts, accent);
   378→    addMerged(darkParts, dark);
   379→
   380→    // ---- headlights + taillights (always-on party-racer look) -----------------
   381→    const lightGeos: THREE.BufferGeometry[] = [];
   382→    for (const s of [-1, 1]) {
   383→      lightGeos.push(T(new THREE.CylinderGeometry(0.085, 0.105, 0.1, 8), s * 0.3, 0.45, 2.14, Math.PI / 2));
   384→    }
   385→    addMerged(lightGeos, new THREE.MeshBasicMaterial({ color: 0xfff6d8 }));
   386→    const tailGeos: THREE.BufferGeometry[] = [];
   387→    for (const s of [-1, 1]) {
   388→      tailGeos.push(T(new THREE.BoxGeometry(0.22, 0.1, 0.06), s * 0.38, 0.62, -1.98));
   389→    }
   390→    addMerged(tailGeos, new THREE.MeshBasicMaterial({ color: 0xff3822 }));
   391→
   392→    // ---- racing roundel decal lying on the sloped nose cone -------------------
   393→    const roundel = new THREE.Mesh(new THREE.CircleGeometry(0.24, 18),
   394→      new THREE.MeshBasicMaterial({ map: getRoundelTexture(), transparent: true }));
   395→    roundel.position.set(0, 0.83, 1.52);
   396→    roundel.rotation.x = -Math.PI / 2 + 0.36;   // follows the nose cone slope
   397→    this.body.add(roundel);
   398→
   399→    // ---- exhaust tips (metal, emissive when boosting) -------------------------
   400→    const tipMat = new THREE.MeshLambertMaterial({ color: 0x8a8a96, emissive: 0x220000 });
   401→    for (const s of [-1, 1]) {
   402→      const tip = new THREE.Mesh(new THREE.CylinderGeometry(0.1, 0.12, 0.22, 7), tipMat);
   403→      tip.rotation.x = Math.PI / 2;
   404→      tip.position.set(s * 0.32, 0.56, -1.85);
   405→      this.body.add(tip);
   406→      this.exhausts.push(tip);
   407→
   408→      const flame = new THREE.Mesh(new THREE.ConeGeometry(0.17, 1.0, 6),
   409→        new THREE.MeshBasicMaterial({ color: 0xffa53a, transparent: true, opacity: 0.9 }));
   410→      flame.rotation.x = Math.PI / 2;
   411→      flame.position.set(s * 0.32, 0.56, -2.4);
   412→      flame.visible = false;
   413→      this.body.add(flame);
   414→      this.boostFlames.push(flame);
   415→    }
   416→    this.materials.push(tipMat);
   417→
   418→    // ---- wheels: fat rear + nimble front, hub with 5 spokes --------------------
   419→    this.buildProceduralWheels();
   420→  }

   424→  private buildCharacter(id: string): void {
   425→    const c = CHARACTER_MAP[id];
   426→    const skinMat = new THREE.MeshLambertMaterial({ color: c.accent });

   894→  /** Landing impulse for the suspension spring. */
   895→  land(strength: number): void { this.bounceVel = -Math.min(1.4, strength * 0.08); }
   896→
   897→  update(dt: number, t: number, v: VisualState): void {
   898→    this.animT += dt;
   899→
   900→    // suspension spring (visual)
   901→    this.bounceVel += (-this.bounce * 90 - this.bounceVel * 12) * dt;
   902→    this.bounce += this.bounceVel * dt;
   903→    if (this.wasAirborne && v.grounded) { this.bounceVel = -1.2; }
   904→    this.wasAirborne = v.airborne;
   905→
   906→    // wheels
   907→    const spin = v.speedRatio * 22 * dt;
   908→    for (const w of this.wheels) w.rotation.x -= spin;
   909→    // NOTE steering sign: positive steer = right on screen = yaw decreases,
   910→    // so the front wheels visually counter-rotate to point into the turn.
   911→    for (const f of this.frontWheels) f.rotation.y = -v.steer * 0.38;
   912→    // GLB-detected wheels: same animation, applied to model nodes
   913→    for (const w of this.glbWheels) w.rotation.x -= spin;
   914→    for (const f of this.glbFrontWheels) {
   915→      f.rotation.y = (this.glbSteerBase.get(f) ?? 0) - v.steer * 0.38;
   916→    }
   917→
   918→    // body roll + drift lean + idle bob + stunt rotation
   919→    const driftLean = v.drifting ? 0.16 : 0;
   920→    // stunt: full 360° over the trick progress (flip or barrel roll)
   921→    let trickRollZ = 0, trickPitchX = 0;
   922→    if (v.trickSpin > 0) {
   923→      const ang = v.trickSpin * Math.PI * 2;
   924→      if (v.trickKind === 0) trickPitchX = -ang;                 // front flip
   925→      else trickRollZ = (v.trickKind === 1 ? 1 : -1) * ang;      // barrel roll
   926→    }
   927→    this.body.rotation.z = THREE.MathUtils.lerp(this.body.rotation.z, -v.steer * (0.055 + driftLean), 0.5) + trickRollZ;
   928→    this.body.position.y = this.bounce + (v.grounded ? Math.sin(t * 9) * 0.012 * (0.3 + v.speedRatio) : 0);
   929→    this.body.rotation.x = (v.boost ? -0.03 : 0.01) + trickPitchX;
   930→
   931→    // character animation
   932→    const lean = v.steer * 0.22 + (v.drifting ? 0.3 : 0);
   933→    this.torso.rotation.z = THREE.MathUtils.lerp(this.torso.rotation.z, -lean, dt * 8);
   934→    this.torso.rotation.x = v.boost ? -0.12 : 0.05 + Math.sin(t * 4) * 0.02;
   935→    this.head.rotation.z = -lean * 0.5;
   936→    this.head.rotation.y = lean * 0.8;
   937→    // hands follow the wheel through the turn (subtle but sells the drive)
   938→    const wheelTurn = v.steer * 0.3 + (v.drifting ? Math.sign(v.steer || 1) * 0.18 : 0);
   939→    this.armL.rotation.y = THREE.MathUtils.lerp(this.armL.rotation.y, wheelTurn, dt * 10);
   940→    this.armR.rotation.y = THREE.MathUtils.lerp(this.armR.rotation.y, wheelTurn, dt * 10);
   941→    // chin tuck at full boost, alert bob while drifting
   942→    this.head.rotation.x = (v.boost ? 0.14 : 0) + Math.sin(t * (v.drifting ? 11 : 3.2)) * (v.drifting ? 0.05 : 0.02);
   943→
   944→    if (this.anim === 'victory') {
   945→      this.armL.rotation.z = THREE.MathUtils.lerp(this.armL.rotation.z, 2.6, dt * 6);
   946→      this.armR.rotation.z = THREE.MathUtils.lerp(this.armR.rotation.z, -2.6, dt * 6);
   947→      this.characterRoot.position.y = this.seatY + Math.abs(Math.sin(this.animT * 6)) * 0.22;
   948→      this.head.rotation.x = -0.3;
   949→    } else if (this.anim === 'defeat') {
   950→      this.armL.rotation.z = THREE.MathUtils.lerp(this.armL.rotation.z, 0.9, dt * 4);
   951→      this.armR.rotation.z = THREE.MathUtils.lerp(this.armR.rotation.z, -0.9, dt * 4);
   952→      this.torso.rotation.x = 0.35;
   953→      this.head.rotation.x = 0.5;
   954→    } else if (this.anim === 'hit') {
   955→      const shake = Math.sin(this.animT * 40) * 0.12 * Math.max(0, 1 - this.animT * 2);
   956→      this.head.rotation.z += shake;
   957→      this.torso.rotation.z += shake * 0.6;
   958→      if (this.animT > 0.5) this.anim = 'idle';
   959→    } else {
   960→      this.armL.rotation.z = THREE.MathUtils.lerp(this.armL.rotation.z, 0.1, dt * 8);
   961→      this.armR.rotation.z = THREE.MathUtils.lerp(this.armR.rotation.z, -0.1, dt * 8);
   962→      this.characterRoot.position.y = this.seatY;
   963→    }
   964→
   965→    // boost flames
   966→    const flameOn = v.boost;
   967→    for (const f of this.boostFlames) {
   968→      f.visible = flameOn;
   969→      if (flameOn) {
   970→        f.scale.set(1 + Math.sin(t * 30) * 0.25, 1 + Math.sin(t * 24) * 0.3, 1);
   971→        (f.material as THREE.MeshBasicMaterial).color.setHSL(0.08 + Math.sin(t * 20) * 0.02, 1, 0.6);
   972→      }
   973→    }
   974→    for (const e of this.exhausts) {
   975→      (e.material as THREE.MeshLambertMaterial).emissive.setHex(flameOn ? 0xaa5500 : 0x220000);
   976→    }
   977→
   978→    // hit flash tint
   979→    const flash = this.anim === 'hit' ? 0.7 * Math.max(0, 1 - this.animT * 2) : 0;
   980→    for (const m of this.materials) {
   981→      if (!(m as any).__baseEmissive) (m as any).__baseEmissive = (m.emissive.getHex?.() ?? 0);
   982→      m.emissive?.setHex((m as any).__baseEmissive | ((flash > 0) ? 0x662222 : 0));
   983→    }
   984→  }
   985→

   987→    this.group.traverse(o => {
   988→      const m = o as THREE.Mesh;
   989→      if (m.geometry) m.geometry.dispose();
   990→    });
   991→  }
